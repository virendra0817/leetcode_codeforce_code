// ===================================================
// Problem  : Minimum Common Value
// Platform : Leetcode
// Link     : https://leetcode.com/problems/minimum-common-value/
// Difficulty: Easy
// Language : python3
// Runtime  : 9 ms
// Memory   : 37.8 MB
// Date     : May 20, 2026
// ===================================================class Solution:
    def getCommon(self, nums1: List[int], nums2: List[int]) -> int:
        i,j = 0,0
        while i < len(nums1) and j <len(nums2):
            if nums1[i] == nums2[j]:
                return nums1[i]
            if nums1[i]< nums2[j]:
                i = i+1
            else:
                j = j+1
        return -1
