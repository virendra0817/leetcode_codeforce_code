// ===================================================
// Problem  : Minimum Moves to Equal Array Elements III
// Platform : Leetcode
// Link     : https://leetcode.com/problems/minimum-moves-to-equal-array-elements-iii/
// Difficulty: Easy
// Language : python3
// Runtime  : 0 ms
// Memory   : 19.1 MB
// Date     : November 4, 2026
// ===================================================class Solution:
    def minMoves(self, nums: List[int]) -> int:
        s = 0 
        t= max(nums)
        for i in nums:
            s  = s + (t-i)
        
        return s
