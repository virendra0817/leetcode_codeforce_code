// Problem  : Move Zeroes
// Platform : Leetcode
// Link     : https://leetcode.com/problems/move-zeroes/
// Difficulty: Easy
// Language : python3
// Runtime  : 6 ms
// Memory   : 20.3 MB
// Date     : May 5, 2026
// ===================================================class Solution:
    def moveZeroes(self, nums: List[int]) -> None:
        """
        Do not return anything, modify nums in-place instead.
        """
        a = []

        for i in range(len(nums) - 1, -1, -1):
            if nums[i] != 0:
              a.append(nums.pop(i))

        a.reverse()
        a.extend(nums)
        nums[:]=a
