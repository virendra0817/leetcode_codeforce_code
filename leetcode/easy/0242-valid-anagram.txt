    def isAnagram(self, s: str, t: str) -> bool:
        if len(s) != len(t):
            return False
        s1=set()
        for i in s :
            s1.add(i)
        
        for i in s1 :
            if t.count(i) != s.count(i):
                return False
        
        return True
