// Problem  : Two Sum
// Platform : Leetcode
// Link     : https://leetcode.com/problems/two-sum/
// Difficulty: Easy
// Language : python3
// Runtime  : 5389 ms
// Memory   : 125.1 MB
// Date     : May 28, 2026
// ===================================================class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        d={}
        for i in range(0,len(nums)):
            for j in range(i+1,len(nums)):
                d[nums[i]+nums[j]] = [i,j]
        
        for j in d.keys():
            if j == target :
                return d[j]
