// ===================================================
// Problem  : Smallest Missing Multiple of K
// Platform : Leetcode
// Link     : https://leetcode.com/problems/smallest-missing-multiple-of-k/
// Difficulty: Easy
// Language : python3
// Runtime  : 0 ms
// Memory   : 19.3 MB
// Date     : August 30, 2026
// ===================================================class Solution:
    def missingMultiple(self, nums: List[int], k: int) -> int:
        l= len(nums)
        f=0
        for i in range(1,l+2):
            if k*i  not in nums:
                f=  k*i
                break
        return f
