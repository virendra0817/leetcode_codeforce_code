// ===================================================
// Problem  : Minimum Element After Replacement With Digit Sum
// Platform : Leetcode
// Link     : https://leetcode.com/problems/minimum-element-after-replacement-with-digit-sum/
// Difficulty: Easy
// Language : python3
// Runtime  : 7 ms
// Memory   : 19.2 MB
// Date     : May 30, 2026
// ===================================================class Solution:
    def minElement(self, nums: List[int]) -> int:
        
        for i in range(len(nums)):
            t=str(nums[i])
            c=0
            for j in t :
                c= c + int(j)
            nums[i] = c 
        return min(nums)
