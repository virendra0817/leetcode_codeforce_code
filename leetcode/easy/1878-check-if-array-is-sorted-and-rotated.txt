// ===================================================
// Problem  : Check if Array Is Sorted and Rotated
// Platform : Leetcode
// Link     : https://leetcode.com/problems/check-if-array-is-sorted-and-rotated/
// Difficulty: Easy
// Language : python3
// Runtime  : 0 ms
// Memory   : 19.3 MB
// Date     : May 23, 2026
// ===================================================class Solution:
    def check(self, nums: List[int]) -> bool:
        c= 0
        n=len(nums)
        for i  in range(len(nums)):
            if nums[i] >  nums[(i+1)%n]:
                c+=1
                if c >1 :
                 return False
        return True
