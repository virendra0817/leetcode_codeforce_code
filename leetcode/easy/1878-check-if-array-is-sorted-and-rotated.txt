// ===================================================
// Problem  : Check if Array Is Sorted and Rotated
// Platform : Leetcode
// Link     : https://leetcode.com/problems/check-if-array-is-sorted-and-rotated/
// Difficulty: Easy
// Language : python3
// Runtime  : 0 ms
// Memory   : 19.3 MB
// Date     : May 20, 2026
// ===================================================class Solution:
    def check(self, nums: List[int]) -> bool:
        count , n= 0, len(nums)
        for i in range(n):
            if nums[i] > nums[(i+1)% n] :
                count+=1
            
                if count >1 :
                  return False

        return True
