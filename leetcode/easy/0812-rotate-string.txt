// Problem  : Rotate String
// Platform : Leetcode
// Link     : https://leetcode.com/problems/rotate-string/
// Difficulty: Easy
// Language : python3
// Runtime  : 0 ms
// Memory   : 19.2 MB
// Date     : May 3, 2026
// ===================================================class Solution:
    def rotateString(self, s: str, goal: str) -> bool:
        for i in range(len(s)):
            a= s[i::]+s[:i:] 
            if a == goal :
                return True
                break
        return False
