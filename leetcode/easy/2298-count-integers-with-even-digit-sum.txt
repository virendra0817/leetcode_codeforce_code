// ===================================================
// Problem  : Count Integers With Even Digit Sum
// Platform : Leetcode
// Link     : https://leetcode.com/problems/count-integers-with-even-digit-sum/
// Difficulty: Easy
// Language : python3
// Runtime  : 9 ms
// Memory   : 19.3 MB
// Date     : May 11, 2026
// ===================================================class Solution:
    def countEven(self, num: int) -> int:
        c=0
        for i in range(1,num+1):
            sui = 0 
            for j in  str(i):
                sui+= int(j)

            if sui%2 ==0 :
                c+=1

        return c
