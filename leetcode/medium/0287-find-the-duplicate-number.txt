// ===================================================
// Problem  : Find the Duplicate Number
// Platform : Leetcode
// Link     : https://leetcode.com/problems/find-the-duplicate-number/
// Difficulty: Medium
// Language : python3
// Runtime  : 27 ms
// Memory   : 33.8 MB
// Date     : May 20, 2026
// ===================================================class Solution:
    def findDuplicate(self, nums: List[int]) -> int:
        s= set()
        for num in nums:
            if num in s :
                return num
            s.add(num)
